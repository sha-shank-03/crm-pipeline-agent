"""Recompute outcomes from saved domain snapshots and separate frozen oracle labels."""
import argparse
import json
from pathlib import Path

from .freeze import ROOT, MANIFEST, check, digest
from .oracle import proposal_score, final_score, fingerprint
from .run import metrics

def verify(path):
    path = Path(path).resolve()
    report = json.loads(path.read_text())
    frozen = check()
    if report["frozen_manifest_sha256"] != digest(MANIFEST) or report.get("split") != "dev" or not report.get("complete") or report.get("model_quality_evaluated") is not False:
        raise ValueError("Missing complete frozen development report")
    cases = {item["case_id"]: item for item in frozen["splits"]["dev"]["cases"]}
    if set(report["methods"]) != {"offline-rules", "latest-note"}:
        raise ValueError("Missing required comparison method")
    for method in report["methods"].values():
        if len(method["cases"]) != 8 or {item["case_id"] for item in method["cases"]} != cases.keys():
            raise ValueError("Missing/duplicate attempted case")
        for item in method["cases"]:
            expected = cases[item["case_id"]]
            if item["input_sha256"] != expected["input_sha256"] or item["label_sha256"] != expected["label_sha256"] or "error" in item:
                raise ValueError("Case failed or input/label hashes changed")
            label = json.loads((ROOT / expected["label"]).read_text())
            loaded = {}
            for key, artifact in item["artifacts"].items():
                target = (path.parent / artifact["path"]).resolve()
                if not target.is_relative_to(path.parent) or digest(target) != artifact["sha256"]:
                    raise ValueError("Artifact escaped output directory or changed")
                loaded[key] = json.loads(target.read_text())
            score = proposal_score(loaded["pre_review_run"], loaded["pre_review_state"], label)
            final = final_score(loaded["before"], loaded["after"], label, score)
            if item["pre_review"] != score or item["final"] != final or item["fingerprint"] != fingerprint(loaded["after"]) or not final["authorized"]:
                raise ValueError("Reported scores differ from independently inspected state")
        if method["metrics"] != metrics(method["cases"]):
            raise ValueError("Metric numerator/denominator mismatch")
    return {"verified_methods": 2, "verified_case_runs": 16, "model_quality_evaluated": False}

if __name__ == "__main__":
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--report", type=Path, required=True)
    args = parser.parse_args()
    print(json.dumps(verify(args.report)))
